# ARIA sidecar — runtime dependencies
# Python 3.11 (BUILD_SPEC §4 — not 3.12; pywinauto/ctranslate2 wheels are safest on 3.11)
#
# This file grows one phase at a time. Every pin below matches BUILD_SPEC §4 exactly.
# Installing all of §4 up front pulls ~1.5GB and risks a Phase 0 failure on a
# Phase 2 dependency, so each phase adds its own lines when it needs them.

# ── Phase 0: foundation ───────────────────────────────────────────
fastapi==0.115.*
uvicorn[standard]==0.32.*
pydantic==2.9.*
pydantic-settings==2.6.*      # §5 names it for config.py; absent from §4's list
httpx==0.27.*
sqlite-vec==0.1.6
structlog==24.4.*

# ── Phase 1.5: multi-provider cloud ───────────────────────────────
# API keys live in Windows Credential Manager (§11), never .env.
# §4 lists none of these. OpenAI and Gemini are reached over httpx rather than
# their vendor SDKs, so `openai` and `google-genai` are deliberately absent —
# two fewer dependency trees to survive PyInstaller (§2.3).
keyring==25.7.*

# ── Phase 2 stage 1: speech out ───────────────────────────────────
# CPU only (rule 2 — the 6GB card holds the language model alone) and torch-free
# (rule 3): kokoro-onnx runs on onnxruntime.
#
# Two of §4's pins do not work and are deliberately not used. kokoro-onnx
# requires onnxruntime>=1.20.1 where §4 says 1.19.*, and numpy>=2.0.2 where §4's
# Phase 4 block says 1.26.*. Both are transitive, so they are left unpinned here
# rather than forced to a version that cannot resolve.
#
# Model weights (~340MB) are not vendored: `data/models/kokoro-v1.0.onnx` and
# `voices-v1.0.bin` from the kokoro-onnx releases page. Missing weights log a
# warning and disable voice; they never stop the sidecar.
kokoro-onnx==0.4.*

# ── Phase 2 stage 2: speech in ────────────────────────────────────
# faster-whisper runs on ctranslate2, not torch (rule 3), and is pinned to CPU
# int8 in providers/stt.py so it never touches the card (rule 2).
#
# 1.0.3 imports `requests` without declaring it, so a clean install of just
# faster-whisper fails at import time. It is listed explicitly below rather than
# left to chance — PyInstaller (§9) would otherwise miss it too.
#
# Weights (~150MB for base.en) download on first use into data/models/whisper/.
# A first launch with no network leaves speech-in disabled, never broken.
faster-whisper==1.0.3
requests>=2.32                # undeclared dependency of faster-whisper 1.0.3

# ── Phase 2 stage 3: hands free ───────────────────────────────────
# openWakeWord's base install is torch-free — torch appears only in its
# [full] extra, which trains new wake words. It runs on ONNX here because
# its tflite-runtime dependency is marked Linux-only.
#
# **webrtcvad, which §4 names, is deliberately absent.** It ships source-only
# and its C extension needs MSVC; it does not build on this machine, and a
# release build would need the toolchain too. faster-whisper already bundles
# Silero VAD as ONNX — the same model stage 2's vad_filter runs — so
# providers/vad.py uses that instead: no new dependency, no compiler, and a
# neural detector rather than webrtcvad's GMM.
#
# Weights (~3.5MB) are not vendored: `python scripts/fetch_wakeword.py`.
openwakeword==0.6.0

# ── Phase 3: tools and Windows control ────────────────────────────
# win32gui enumerates top-level windows for `list_windows`; psutil reads the
# machine for `get_system_info`.
#
# pycaw is **not** in §4 and is added deliberately: setting an absolute system
# volume needs the IAudioEndpointVolume COM interface, and there is no way to
# reach it from the standard library. It pulls comtypes and nothing else — no
# torch, no build step. Note its API changed: `AudioUtilities.GetSpeakers()`
# returns a wrapper with `.EndpointVolume`, not a raw COM object, so every
# `Activate(IID, CLSCTX_ALL, None)` example on the internet now fails.
#
# pywinauto (§4) is deferred again: it is only needed for `focus_window` and
# `close_app`, which land with the remaining tools.
pywin32==306
psutil==6.0.*
pycaw==20251023

# ── Phase 4: the finder ───────────────────────────────────────────
# Text extraction for the semantic index. watchfiles is already present via
# uvicorn[standard] at 1.2.0 rather than §4's 0.24 pin — pinning it down while
# the sidecar holds its .pyd open fails on Windows, and the API used here is
# unchanged between them.
#
# Embeddings come from Ollama over httpx (`nomic-embed-text`, ~274MB, pulled
# separately), so there is no embedding library here and no torch.
pypdf==5.0.*
python-docx==1.1.*
openpyxl==3.1.*
